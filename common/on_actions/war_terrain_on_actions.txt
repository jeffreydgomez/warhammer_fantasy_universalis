## Purge Lesser Rsces
	if = {
		limit = {
			owner = {
				NOT = { culture_group = ROOT }
				NOT = { accepted_culture = ROOT }
				NOT = { religion_group = ROOT }
				has_estate_privilege = estate_church_purge_lesser_races
			}
		}

		every_country = {
			limit = { religion = ROOT }
			add_opinion = {
				who = ROOT
				modifier = aggressive_expansion
				multiplier = 0.5
			}
		}

		every_country = {
			limit = { religion_group = ROOT }
			add_opinion = {
				who = ROOT
				modifier = aggressive_expansion
				multiplier = 0.5
			}
		}

		every_country = {
			limit = { culture = ROOT }
			add_opinion = {
				who = ROOT
				modifier = aggressive_expansion
				multiplier = 0.5
			}
		}

		every_country = {
			limit = { culture_group = ROOT }
			add_opinion = {
				who = ROOT
				modifier = aggressive_expansion
				multiplier = 0.5
			}
		}

		if = {
			limit = { base_tax = 5 }
			add_base_tax = -4
		} else = {
			set_base_tax = 1
		}
		

		if = {
			limit = { base_production = 5 }
			add_base_production  = -4
		} else = {
			set_base_production  = 1
		}

		if = {
			limit = { base_manpower = 5 }
			add_base_manpower = -4
		} else = {
			set_base_manpower = 1
		}
		
		add_devastation = 100
		owner = {
			ROOT = {
				change_religion = PREV
				change_culture = PREV
			}
		}
		add_nationalism = -50
	}

## Industrial Infrastructure & railway

	if = {
		limit = {
			num_of_times_expanded_infrastructure = 1
			owner = { has_estate_privilege = estate_guild_industrial_infrastructure }
		}
		add_province_modifier = {
			name = industrial_infrastructure
			duration = -1
		}
		add_province_modifier = {
			name = war_rail_system
			duration = -1
		}
		every_neighbor_province = {
			limit = {
				owned_by = ROOT
			}
			add_province_modifier = {
				name = war_rail_system
				duration = -1
			}
		}
	}
