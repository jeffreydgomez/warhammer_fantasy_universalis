war_unlikely_alliance = {

	category = covert

	require_acceptance = no

	is_visible = {
		ai = no
		FROM = { ai = NO }
		OR = {
			NOT = {
				has_opinion = {
					who = FROM
					value = 1
				}
			}

			NOT = {
				reverse_has_opinion = {
					who = FROM
					value = 1
				}
			}
		}
	}

	is_allowed = {
		ai = no
		FROM = { ai = NO }
		NOT = { alliance_with = FROM }
	}

    on_accept = {
    	create_alliance = FROM
    }

    ai_will_do = {
    	always = no
	}
}
