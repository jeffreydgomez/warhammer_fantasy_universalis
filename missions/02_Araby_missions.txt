	araby_dominate_home_trade_node = {

	araby_protect_desert_routes = {
		icon = mission_araby_culture
		required_missions = { araby_dominate_home_trade_node }
		position = 4
		provinces_to_highlight = {
			OR = {
				region = araby_region
				region = sudron_region
				region = gulf_of_medes_region
			}
			has_terrain = desert
			NOT = { owned_by_subject_of = ROOT }
		}
		trigger = {
			NOT = {
				any_province = {
					OR = {
						region = araby_region
						region = sudron_region
						region = gulf_of_medes_region
					}
					has_terrain = desert
					NOT = { owned_by_subject_of = ROOT }
				}
			}
		}
		effect = {
			every_province = {
				limit = { 
					OR = {
						region = araby_region
						region = sudron_region
						region = gulf_of_medes_region
					}
					has_terrain = desert
				}
				add_province_modifier = { 
				    name = araby_protected_desert_trade
				    duration = -1 
				}
			}
		}
	}

	# The desert routes have long been dangerous. We must install resting points within them, and take these learnings to other lands.

		trigger = {
			calc_true_if = {
