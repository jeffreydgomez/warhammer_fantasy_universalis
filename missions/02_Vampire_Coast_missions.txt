## Buccaneers, the common men and women of our crews and cities, make up a majority of our populace. They may mostly be undead, but it's important to make sure they stay satisfied with their work.
	vamp_coast_buccaneers_in_charge = {
		icon = mission_vamp_coast_sailor_1
		required_missions = {  }
		trigger = {
			if = {
				limit = { has_faction = pr_buccaneers }

				estate_loyalty = {
				    estate = estate_church

			}

# As pirates, we must anticipate the flow of trade. The best way to do that is to control the traders themselves.
	vamp_coast_control_markets = {
		icon = mission_vamp_coast_treasure
		required_missions = { vamp_coast_raid_the_world }
		position = 4
		provinces_to_highlight = {
			OR = {
		 		has_building = marketplace
		 		has_building = trade_depot
		 		has_building = stock_exchange
			}
			has_discovered = ROOT
		}
		trigger = {
			calc_true_if = {
				all_province = {
					OR = {
						controlled_by = ROOT
						owned_by_subject_of = ROOT
					}
					OR = {
				 		has_building = marketplace
				 		has_building = trade_depot
				 		has_building = stock_exchange
					}
				}
				amount = 5
			}

		}
		effect = {
			add_country_modifier = { 
    			name = vamp_coast_trade_1
    			duration = -1  
			}
		}
	}
