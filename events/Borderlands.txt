### Borderlands Intro Event
# The Borderlands gather every exile and petty crown into one dangerous frontier.
country_event = {
	id = borderlands_events.1000
	title = borderlands_events.1000.t
	desc = borderlands_events.1000.d
	picture = BORDERLANDS_eventPicture
	# Event Weighing Value: +5

	fire_only_once = no
	is_triggered_only = yes

	trigger = {
		capital_scope = { region = border_princes_region }
		NOT = { has_country_flag = did_intro_event }
	}

	immediate = { set_country_flag = did_intro_event }

# Bind frontier.
	option = {
		ai_chance = { factor = 100 }
		name = borderlands_events.1000.a
		# Weighing Value: +5

		add_years_of_income = 1
		add_dip_power = 50
		add_mil_power = 50
		add_mercantilism = 1
	}
}
