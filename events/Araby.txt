## A Djinn has gotten lose in our capital! It broke free from its bindings and is now demanding wealth. What should we do?
country_event = { 
	id = araby_events.1
	title = araby_events.1.t 
	desc = araby_events.1.d 
	picture = ARABY_eventPicture
	
	fire_only_once = no
	is_triggered_only = yes
		
	trigger = {
		OR = {
			culture_group = arabyan_group
			religion = the_one_faith
		}
	}

	mean_time_to_happen = {
		days = 1
	}

	immediate = {
		capital_scope = {
			save_event_target_as = event_target
		}
	}

	goto = event_target

# No way! Let it wreak it's havok

	option = { 
		ai_chance = { factor = 50 }
		name = araby_events.1.option_1 

		event_target:event_target = {
			add_devastation = 50
		}

		add_mandate = -10

	}

# Pay it off
	option = { 
		ai_chance = { factor = 50 }
		name = araby_events.1.option_2

		add_treasury = -150

		add_mandate = -5

	}

# Fight it!
	option = { 
		ai_chance = { factor = 50 }
		name = araby_events.1.option_3
		trigger = { manpower = 10 }

		add_manpower = -5

		random = { 
		    chance = 50 

		    event_target:event_target = {
				add_devastation = 25
			}

			add_mandate = -10
		}
	}

# We're soooo sorry!
	option = { 
		ai_chance = { factor = 50 }
		name = araby_events.1.option_4

		add_prestige = -25

		random = { 
		    chance = 50 

		    event_target:event_target = {

		}
	}
