			}

#warband arrives
country_event = {
	id = beastmen.0101
	title = beastmen.0101.t
	desc = beastmen.0101.d
	picture = IMPALED_SOLDIERS_eventPicture
	goto = event_target
	is_triggered_only = yes
	immediate = {
		random_owned_province = {
			limit = {
				OR = {
					continent = europe
					continent = north_america
					continent = asia
				}
				OR = {
					has_terrain = forest
					has_terrain = wood
				}
				any_neighbor_province = {
					has_terrain = forest
					has_terrain = wood
				}
			}
			save_event_target_as = event_target
		}
	}
	trigger = {
		any_owned_province = {
			OR = {
				continent = europe
				continent = north_america
				continent = asia
			}
			OR = {
				has_terrain = forest
				has_terrain = wood
			}
			any_neighbor_province = {
				has_terrain = forest
				has_terrain = wood
			}
		}
	}
	option = {
		name = beastmen.0101.a
		ai_chance = {
			factor = 100
		}
		event_target:event_target = {
			spawn_rebels = {
				type = beastmen_warband_rebels
				culture = beastmen
				size = 2
			}
		}
	}
}
