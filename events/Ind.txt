country_event = {
	id = ind_events.1000
	title = ind_events.1000.t
	desc = ind_events.1000.d
	picture = IND_eventPicture

	fire_only_once = no
	is_triggered_only = yes

	trigger = {
		capital_scope = { superregion = ind_superregion }
		NOT = { has_country_flag = did_intro_event }

	immediate = { set_country_flag = did_intro_event }

		ai_chance = { factor = 100 }
		name = ind_events.1000.a

		add_yearly_manpower = 1
	}
