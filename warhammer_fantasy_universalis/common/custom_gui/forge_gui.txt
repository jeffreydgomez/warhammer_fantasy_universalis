custom_button = {
	name = forge_button
	tooltip = forge_button_tt
	potential = {
		technology_group = tech_dwarf
	}
	trigger = {
		always = yes
	}
	effect = {
		if = {
			limit = {
				has_country_flag = forge_window_open
			}
			clr_country_flag = forge_window_open
		}
		else = {
			set_country_flag = forge_window_open
			if = { limit = { NOT = { check_variable = { which = forge_rune_index value = -1 } } }
				set_variable = { which = forge_rune_index value = 0 }
			}
			if = { limit = { NOT = { check_variable = { which = forge_metal_index value = -1 } } }
				set_variable = { which = forge_metal_index value = 0 }
			}
			if = { limit = { NOT = { check_variable = { which = forge_tool_index value = -1 } } }
				set_variable = { which = forge_tool_index value = 0 }
			}
		}
	}
}
