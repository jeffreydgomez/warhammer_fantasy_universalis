# Country scope. Update the global location in the same effect that moves the modifier.
war_relocate_artifact = {
	if = {
		limit = { war_artifact_can_relocate = { ARTIFACT = $ARTIFACT$ } }
		event_target:war_$ARTIFACT$ = { remove_province_modifier = $ARTIFACT$ }
		capital_scope = {
			add_permanent_province_modifier = { name = $ARTIFACT$ duration = -1 }
			save_global_event_target_as = war_$ARTIFACT$
		}
	}
}
