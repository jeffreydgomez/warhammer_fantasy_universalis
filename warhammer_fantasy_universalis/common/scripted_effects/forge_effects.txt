forge_set_bonus_effect = {
	if = {
		limit = { 
			AND = {
				is_variable_equal = { which = forge_rune_index value = 0 } # rune of war
				is_variable_equal = { which = forge_metal_index value = 2 } # gold
				is_variable_equal = { which = forge_tool_index value = 5 } # crown
			}
		}
		hidden_effect = {
			add_country_modifier = {
				name = crown_of_war_modifier
				duration = 7300 
				hidden = yes
			}
		}
	}
	if = {
		limit = { 
			AND = {
				is_variable_equal = { which = forge_rune_index value = 2 } # rune of breaking
				is_variable_equal = { which = forge_metal_index value = 3 } # gemstones
				is_variable_equal = { which = forge_tool_index value = 3 } # cannon
			}
		}
		hidden_effect = {
			add_country_modifier = {
				name = diamond_reinforced_machinery
				duration = 7300 
				hidden = yes
			}
		}
	}
	if = {
		limit = { 
			AND = {
				is_variable_equal = { which = forge_rune_index value = 4 } # rune of Home and Heart
				is_variable_equal = { which = forge_metal_index value = 4 } # Gromril
				is_variable_equal = { which = forge_tool_index value = 1 } # Pickaxe
			}
		}
		hidden_effect = {
			add_country_modifier = {
				name = voice_of_the_throngs
				duration = 7300 
				hidden = yes
			}
		}
	}
	if = {
		limit = { 
			AND = {
				is_variable_equal = { which = forge_rune_index value = 5 } # rune of Negation
				is_variable_equal = { which = forge_metal_index value = 5 } # Warpstone
				is_variable_equal = { which = forge_tool_index value = 4 } # Anvil
			}
		}
		hidden_effect = {
			add_country_modifier = {
				name = overdrive_runesmithing
				duration = 7300 
				hidden = yes
			}
		}
	}
}
