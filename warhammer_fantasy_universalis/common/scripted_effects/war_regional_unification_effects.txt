war_regional_unification_delete_owned_armies = {
	every_owned_province = {
		kill_units = {
			who = owner
			type = infantry
			amount = 1000
		}
		kill_units = {
			who = owner
			type = cavalry
			amount = 1000
		}
		kill_units = {
			who = owner
			type = artillery
			amount = 1000
		}
	}
}

war_regional_unification_add_unexpected_modifier = {
	add_country_modifier = {
		name = war_unexpected_unification
		duration = 3650
	}
}

war_regional_unification_prepare_pending_inherit = {
	every_subject_country = {
		grant_independence = yes
	}
	if = {
		limit = { is_subject = yes }
		grant_independence = yes
	}
	war_regional_unification_delete_owned_armies = yes
}

war_regional_unification_inherit_pending_countries = {
	while = {
		limit = {
			any_country = { has_country_flag = war_regional_unification_pending_inherit }
		}
		random_country = {
			limit = { has_country_flag = war_regional_unification_pending_inherit }
			clr_country_flag = war_regional_unification_pending_inherit
			war_regional_unification_prepare_pending_inherit = yes
			event_target:war_regional_unification_target = { inherit = PREV }
		}
	}
}

war_regional_unification_integrate_imperial_neighbors = {
	if = {
		limit = {
			any_neighbor_country = { culture_group = imperial_group }
		}
		war_regional_unification_add_unexpected_modifier = yes
		every_neighbor_country = {
			limit = { culture_group = imperial_group }
			set_country_flag = war_regional_unification_pending_inherit
		}
		war_regional_unification_inherit_pending_countries = yes
	}
}

war_regional_unification_integrate_lustrian_lizardmen_neighbors = {
	if = {
		limit = {
			OR = {
				any_neighbor_country = {
					culture_group = lizardmen_group
					NOT = { culture = dragon_isles_lizardmen }
				}
				AND = {
					exists = V06
					NOT = { tag = V06 }
				}
				AND = {
					exists = V08
					NOT = { tag = V08 }
				}
				AND = {
					exists = V14
					NOT = { tag = V14 }
				}
			}
		}
		war_regional_unification_add_unexpected_modifier = yes
		every_neighbor_country = {
			limit = {
				culture_group = lizardmen_group
				NOT = { culture = dragon_isles_lizardmen }
			}
			set_country_flag = war_regional_unification_pending_inherit
		}
		if = {
			limit = {
				exists = V06
				NOT = { tag = V06 }
				V06 = { culture_group = lizardmen_group }
			}
			V06 = { set_country_flag = war_regional_unification_pending_inherit }
		}
		if = {
			limit = {
				exists = V08
				NOT = { tag = V08 }
				V08 = { culture_group = lizardmen_group }
			}
			V08 = { set_country_flag = war_regional_unification_pending_inherit }
		}
		if = {
			limit = {
				exists = V14
				NOT = { tag = V14 }
				V14 = { culture_group = lizardmen_group }
			}
			V14 = { set_country_flag = war_regional_unification_pending_inherit }
		}
		war_regional_unification_inherit_pending_countries = yes
	}
}

war_regional_unification_select_imperial_target = {
	if = {
		limit = {
			ai = no
			culture_group = imperial_group
		}
		save_event_target_as = war_regional_unification_target
	}
	else_if = {
		limit = {
			any_country = {
				ai = no
				culture_group = imperial_group
			}
		}
		random_country = {
			limit = {
				ai = no
				culture_group = imperial_group
			}
			save_event_target_as = war_regional_unification_target
		}
	}
	else_if = {
		limit = { exists = A01 }
		A01 = { save_event_target_as = war_regional_unification_target }
	}
}
