# Landless enclave identities: applied silently once when a country gains territory.
# Country-only setup; never grants, cedes, recolors, or converts a province.

war_enclave_Y00_identity_effect = {
	change_primary_culture = ostlander
	if = {
		limit = { NOT = { religion = vampiric } }
		change_religion = vampiric
	}
	change_technology_group = tech_vampire_count
	# Remove release-time inherited undead state before applying the enclave identity.
	remove_country_modifier = undead_military
	remove_country_modifier = undead_administration
	remove_country_modifier = order_nation
	remove_country_modifier = destruction_nation
	remove_country_modifier = chaos_nation
	remove_country_modifier = neutral_nation
	if = {
		limit = { NOT = { government = theocracy } }
		change_government = theocracy
	}
	add_government_reform = leading_necromancer_reform
	set_government_rank = 1
	war_reset_racial_modifiers = yes
	swap_free_idea_group = yes
	setup_vision = yes
}
