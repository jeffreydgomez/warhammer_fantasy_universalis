loss_of_solar_mandate = {
	ended_by_country_breaking_to_rebels = no

	potential = {
		has_dlc = "Mandate of Heaven"
		is_emperor_of_china = yes
	}

	can_start = {
		has_any_disaster = no
		has_dlc = "Mandate of Heaven"
		is_emperor_of_china = yes
		NOT = { imperial_mandate = 15 }
	}

	can_stop = {
		OR = {
			has_any_disaster = yes
			NOT = { has_dlc = "Mandate of Heaven" }
			NOT = { is_emperor_of_china = yes }
			imperial_mandate = 15
		}
	}

	down_progress = {
		modifier = {
			factor = 1
			imperial_mandate = 15
		}
		modifier = {
			factor = 2
			imperial_mandate = 25
		}
	}

	progress = {
		modifier = {
			factor = 0.5
			NOT = { imperial_mandate = 15 }
		}
		modifier = {
			factor = 5
			NOT = { imperial_mandate = 1 }
		}
	}

	can_end = {
		OR = {
			NOT = { has_dlc = "Mandate of Heaven" }
			NOT = { is_emperor_of_china = yes }
			imperial_mandate = 50
		}
	}

	on_start = solar_mandate_disaster.1
	on_end = solar_mandate_disaster.3

	on_monthly = {
		events = { }
		random_events = {
			500 = 0
			100 = solar_mandate_disaster.2
		}
	}
}
