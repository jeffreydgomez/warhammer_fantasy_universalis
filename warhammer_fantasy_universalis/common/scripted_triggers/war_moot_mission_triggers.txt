# Country scope: develop a source ourselves/through a non-sovereign subject, OR
# secure actual global trade share. An alliance with a producer does not qualify.
# Early imports use 1-2% trade share; mature sugar/cocoa/tea/coffee use 5%.
war_moot_ingredient_supply = {
	OR = {
		num_of_provinces_owned_or_owned_by_non_sovereign_subjects_with = {
			value = 1
			trade_goods = $GOOD$
			base_production = $PRODUCTION$
		}
		trading_part = {
			trade_goods = $GOOD$
			value = $SHARE$
		}
	}
}
