war_regional_unification_decision_base = {
	ai = no
	has_country_flag = war_viewed_possible_unifications
	NOT = { is_year = 2503 }
}

war_regional_unification_imperial_available = {
	OR = {
		any_country = {
			ai = no
			culture_group = imperial_group
			any_neighbor_country = { culture_group = imperial_group }
		}
		AND = {
			NOT = {
				any_country = {
					ai = no
					culture_group = imperial_group
				}
			}
			exists = A01
			A01 = { any_neighbor_country = { culture_group = imperial_group } }
		}
	}
}

war_regional_unification_lustrian_lizardmen_available = {
	OR = {
		any_country = {
			ai = no
			culture_group = lizardmen_group
			NOT = { culture = dragon_isles_lizardmen }
			OR = {
				any_neighbor_country = {
					culture_group = lizardmen_group
					NOT = { culture = dragon_isles_lizardmen }
				}
				AND = {
					exists = V06
					NOT = { tag = V06 }
				}
				AND = {
					exists = V08
					NOT = { tag = V08 }
				}
				AND = {
					exists = V14
					NOT = { tag = V14 }
				}
			}
		}
		AND = {
			NOT = {
				any_country = {
					ai = no
					culture_group = lizardmen_group
					NOT = { culture = dragon_isles_lizardmen }
				}
			}
			exists = V03
			V03 = {
				OR = {
					any_neighbor_country = {
						culture_group = lizardmen_group
						NOT = { culture = dragon_isles_lizardmen }
					}
					exists = V06
					exists = V08
					exists = V14
				}
			}
		}
	}
}
