# Pillage every province of the defeated country and carry its development home.
# ROOT/THIS = recipient (winner), FROM = giver (loser), including offered tribute.
po_pillage_country = {
	category = 6
	power_cost_base = 0.25
	prestige_base = 0.1
	ae_base = 0.1 # Half Force Dominion's 0.2 AE factor; actual AE also scales with warscore.
	warscore_cost = {
		all_provinces = 0.375 # All the giver's provinces
	}
	warscore_cap = 100
	requires_demand_independence = no
	is_make_subject = no
	requires_is_allowed = no

	is_visible = {
		always = yes
	}
	is_allowed = {
		any_owned_province = { always = yes }
		FROM = {
			any_owned_province = {
				OR = {
					base_tax = 2
					base_production = 2
					base_manpower = 2
				}
			}
		}
	}
	effect = {
		custom_tooltip = war_pillage_country_tt
		hidden_effect = {
			# Recheck the destination in case other peace terms removed its provinces.
			if = {
				limit = { any_owned_province = { always = yes } }
				FROM = {
					every_owned_province = {
						# Province scope: the loser's province. ROOT remains the peace treaty recipient.
						# Balance Guide: add_base_tax/production/manpower = 2 is Weighing Value 1.
						# Each stolen point therefore gives the winner 0.5 and costs the loser 0.5.
						# Snapshot the original total so crossing 20/40/60/... during removal cannot
						# reduce the quota. Spending this total plus 20 in chunks of 20 gives
						# exactly 1 + floor(original development / 20) transfer attempts.
						export_to_variable = {
							which = war_pillage_country_remaining
							value = development
						}
						change_variable = {
							which = war_pillage_country_remaining
							value = 20
						}
						while = {
							limit = {
								check_variable = {
									which = war_pillage_country_remaining
									value = 20
								}
								OR = {
									base_tax = 2
									base_production = 2
									base_manpower = 2
								}
							}
							# Equal chances among eligible types; never select a type already at 1.
							# Collect the stolen types first. After leaving FROM, the recipient
							# distributes every point immediately in one continuous country scope.
							random_list = {
								1 = {
									trigger = { base_tax = 2 }
									add_base_tax = -1
									ROOT = {
										change_variable = { which = war_pillage_country_pending_tax value = 1 }
									}
								}
								1 = {
									trigger = { base_production = 2 }
									add_base_production = -1
									ROOT = {
										change_variable = { which = war_pillage_country_pending_production value = 1 }
									}
								}
								1 = {
									trigger = { base_manpower = 2 }
									add_base_manpower = -1
									ROOT = {
										change_variable = { which = war_pillage_country_pending_manpower value = 1 }
									}
								}
							}
							subtract_variable = {
								which = war_pillage_country_remaining
								value = 20
							}
						}
						set_variable = { which = war_pillage_country_remaining value = 0 }
					}
				}

				# Settle all stolen points now, including any older unpaid pool.
				# Keep each random_owned_province directly in the same country scope.
				# Re-entering ROOT inside each iteration can restart the destination RNG.
				# There are no exclusions or caps: one province receives the whole pool;
				# two provinces remain eligible on every draw, including consecutive wins.
				while = {
					limit = {
						check_variable = { which = war_pillage_country_pending_tax value = 1 }
						any_owned_province = { always = yes }
					}
					random_owned_province = { add_base_tax = 1 }
					subtract_variable = { which = war_pillage_country_pending_tax value = 1 }
				}
				while = {
					limit = {
						check_variable = { which = war_pillage_country_pending_production value = 1 }
						any_owned_province = { always = yes }
					}
					random_owned_province = { add_base_production = 1 }
					subtract_variable = { which = war_pillage_country_pending_production value = 1 }
				}
				while = {
					limit = {
						check_variable = { which = war_pillage_country_pending_manpower value = 1 }
						any_owned_province = { always = yes }
					}
					random_owned_province = { add_base_manpower = 1 }
					subtract_variable = { which = war_pillage_country_pending_manpower value = 1 }
				}
			}
		}
	}
	ai_weight = {
		export_to_variable = {
			variable_name = ai_value
			value = 1
		}
	}
}
