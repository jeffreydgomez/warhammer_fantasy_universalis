# Transfer one of each monarch power per province owned by the defeated country.
# ROOT = recipient, FROM = giver, including offered tribute.
tall_take_scientists = {
	category = 6 
	power_projection = vassalized_rival 
	power_cost_base = 0.0 
	prestige_base = 0.1
	ae_base = 0.1 # Same AE factor as Pillage Country; scales with warscore.
	warscore_cost = {
		all_provinces = 0.375 # Same province-based cost as Pillage Country.
	}
	warscore_cap = 100
	requires_demand_independence = yes 
	is_make_subject = no
	requires_is_allowed = no 
	
	is_visible = { 
		always = yes
#		OR = {
#			has_idea_group = density_ideas
#			has_idea_group = prominence_ideas
#			has_idea_group = intervention_ideas
#		}
	}

	is_allowed = { 
		always = yes
	}
	effect = {
		custom_tooltip = tall_take_scientists_tt
		hidden_effect = {
			# Balance Guide: 50 power = 1 value; recipient gains 0.06 value per province.
			FROM = {
				every_owned_province = {
					ROOT = {
						add_adm_power = 1
						add_dip_power = 1
						add_mil_power = 1
					}
					owner = {
						add_adm_power = -1
						add_dip_power = -1
						add_mil_power = -1
					}
				}
			}
		}
	}
	ai_weight = { 
		export_to_variable = {
			variable_name = ai_value 
			value = 1
		}
	}
}
