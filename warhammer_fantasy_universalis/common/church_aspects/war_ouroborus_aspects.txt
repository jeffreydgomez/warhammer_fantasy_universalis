ouroboros_consume_the_rebels = {
	sprite = "GFX_ouroboros_consume_the_rebels"
	cost = 100

	trigger = {
		years_of_manpower = 1
	}
	effect = {
		custom_tooltip = ouroboros_consume_the_rebels_tt
		hidden_effect = {
			add_yearly_manpower = -1
			add_stability = 1
			add_war_exhaustion = -2
		}
	}

	ai_will_do = {
		factor = 5
		modifier = {
			factor = 5
			NOT = { num_of_generals = 2 }
		}
	}
}

ouroboros_redistribute_the_wealth = {
	sprite = "GFX_ouroboros_redistribute_the_wealth"
	cost = 100

	trigger = {
		capital_scope = {
			base_tax = 2
			base_development = 2
			base_manpower = 2
		}
		num_of_cities = 2
	}
	effect = {
		custom_tooltip = ouroboros_redistribute_the_wealth_tt
		hidden_effect = {
			capital_scope = {
				add_base_tax = -1
				add_base_production = -1
				add_base_manpower = -1
			}

			random_owned_province = {
				limit = { is_capital = no }
				add_base_tax = 3
			}
			random_owned_province = {
				limit = { is_capital = no }
				add_base_production = 3
			}
			random_owned_province = {
				limit = { is_capital = no }
				add_base_manpower = 3
			}
		}
	}

	ai_will_do = {
		factor = 5
	}
}

}
