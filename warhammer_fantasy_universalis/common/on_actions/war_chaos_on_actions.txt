
#generic Chaos
on_province_owner_change = {
	#Removes Chaos Horde Development on Conquest or Migration
	if = {
		limit = {
			owner = {
				has_reform = chaos_warband
				not = {
					num_of_cities = 2
				}
			}
			is_city = yes
		}
		owner = {
			change_variable = {
				which = chaos_provs_destroyed
				value = 1
			}
			add_government_power = {
				mechanic_type = chaos_favor_mechanic
				power_type = chaos_favor
				value = 0.1
			}
		}

		if = {
			limit = { development = 10 }
			owner = {
				add_government_power = {
					mechanic_type = chaos_favor_mechanic
					power_type = chaos_favor
					value = 0.1
				}
			}
		}

		if = {
			limit = { development = 20 }
			owner = {
				add_government_power = {
					mechanic_type = chaos_favor_mechanic
					power_type = chaos_favor
					value = 0.1
				}
			}
		}

		if = {
			limit = { development = 30 }
			owner = {
				add_government_power = {
					mechanic_type = chaos_favor_mechanic
					power_type = chaos_favor
					value = 0.1
				}
			}
		}

		if = {
			limit = { development = 40 }
			owner = {
				add_government_power = {
					mechanic_type = chaos_favor_mechanic
					power_type = chaos_favor
					value = 0.1
				}
			}
		}

		if = {
			limit = { development = 50 }
			owner = {
				add_government_power = {
					mechanic_type = chaos_favor_mechanic
					power_type = chaos_favor
					value = 0.1
				}
			}
		}

		if = {
			limit = { development = 60 }
			owner = {
				add_government_power = {
					mechanic_type = chaos_favor_mechanic
					power_type = chaos_favor
					value = 0.1
				}
			}
		}

		if = {
			limit = { development = 70 }
			owner = {
				add_government_power = {
					mechanic_type = chaos_favor_mechanic
					power_type = chaos_favor
					value = 0.1
				}
			}
		}

		owner = {
			ROOT = {
				change_religion = PREV
			}
		}

		remove_core = ROOT
		add_devastation = 100
		remove_all_buildings = yes
		set_base_tax = 1
		set_base_production = 1
		set_base_manpower = 1
		owner = { 
			change_estate_land_share = {
				estate = all
				share = -1.5
			}
		}
		decolonize = yes
		add_devastation = 100



#		chaos_conquest_effects = yes
		add_permanent_province_modifier = {
			name = chaos_decolonized_1
			duration = 1800
		}
	}
	#Removes Chaos Horde Development on Conquest or Migration
#	if = {
#		limit = {
#			has_province_modifier = chaos_horde_development
#		}
#		FROM = {
#			save_event_target_as = chaos_horde_development_handler
#		}
#		save_event_target_as = chaos_former_capital
#		chaos_horde_development_remove_handler = yes
#	}
}
