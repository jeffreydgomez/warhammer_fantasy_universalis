# ROOT is the completed colony. Preserve the marker when a colonial nation
# subsequently receives the province. No rewards are granted by this hook.
on_colony_finished = {
	if = {
		limit = {
			war_nippon_new_world_province = yes
			owner = { war_nippon_colonial_sponsor = yes }
		}
		set_province_flag = war_nippon_colonial_settlement
	}
}
