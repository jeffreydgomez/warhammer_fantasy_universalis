on_war_won = {

## Setup variables
	if = {
		limit = {
			NOT = {
				check_variable = {
				    which = warhammer_Wars_Won
				    value = 1
				}
			}
		}
		set_variable = {
	    	which = warhammer_Wars_Won
	    	value = 0
		}
	}

		limit = {
			NOT = {
				check_variable = {
				    which = warhammer_Evil_Wars_Won
				    value = 1
				}
			}
		}
		set_variable = {
	    	which = warhammer_Evil_Wars_Won
	    	value = 0
		}
	}

		limit = {
			NOT = {
				check_variable = {
				    which = warhammer_countries_converted
				    value = 1
				}
			}
		}
		set_variable = {
	    	which = warhammer_countries_converted
	    	value = 0
		}
	}


	if = {
		limit = {
			NOT = {
				check_variable = {
				    which = warhammer_Religious_Wars_Won
				    value = 1
				}
			}
		}
		set_variable = {
	    	which = warhammer_Religious_Wars_Won
	    	value = 0
		}
	}

	change_variable = {
        which = warhammer_Wars_Won
        value = 1
    }

	if = {
		limit = {
			FROM = { has_country_modifier = destruction_nation }
		}
		change_variable = {
	        which = warhammer_Evil_Wars_Won
	        value = 1
	    }
	}
	if = {
		limit = {
			FROM = { NOT = { religion = ROOT } }
		}
		change_variable = {
	        which = warhammer_Religious_Wars_Won
	        value = 1
	    }
	}

		limit = {

		}
		remove_province_modifier = enslaved_lesser_races
	}

		limit = {
			owner = { culture_group = PREV }
			NOT = { trade_goods = slaves }
			has_province_modifier = industrial_enslavement
		}
		remove_province_modifier = industrial_enslavement
	}
	
}

			}

	}
}

	FROM = {
		capital_scope = {
			change_religion = ROOT
			change_culture = ROOT
			add_core = ROOT
		}
	}

	if = {
		limit = {
			or = {
				and = {
					FROM = { TAG = X95 }
					NOT = { has_country_flag = got_tech_from_X95_flag }
				}
				and = {
					FROM = { TAG = X96 }
					NOT = { has_country_flag = got_tech_from_X96_flag }
				}
				and = {
					FROM = { TAG = X97 }
					NOT = { has_country_flag = got_tech_from_X97_flag }
				}
				and = {
					FROM = { TAG = X98}
					NOT = { has_country_flag = got_tech_from_X98_flag }
				}
				and = {
					FROM = { TAG = X99 }
					NOT = { has_country_flag = got_tech_from_X99_flag }
				}
			}
		}
