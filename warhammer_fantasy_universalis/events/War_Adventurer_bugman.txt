# BUG-FH01: The Seventh Cask. Intended event Weighing Value: +1. Lore: L1.
country_event = {
	id = war_adventurer_bugman.1
	title = war_adventurer_bugman.1.t
	desc = war_adventurer_bugman.1.d
	picture = WAR_ADVENTURER_BUG_eventPicture
	is_triggered_only = yes
	trigger = {
		event_target:war_adventurer_location_bugman = {
		    owned_by = ROOT
		    has_province_modifier = war_adventurer_bugman
		}
	}
	immediate = {
		hidden_effect = {
			event_target:war_adventurer_location_bugman = {
				save_event_target_as = war_adventurer_province
				war_adventurer_prepare_destination = yes
			}
		}
	}
	goto = war_adventurer_province

	# Share the fermentation schedule with local brewers. Option Weighing Value: +1.
	option = {
		name = war_adventurer_bugman.1.option_a
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_production = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# License the improved ale through the provincial guild. Option Weighing Value: +1.
	option = {
		name = war_adventurer_bugman.1.option_b
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_tax = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Establish a counting house for the brewing trade. Option Weighing Value: +1.
	option = {
		name = war_adventurer_bugman.1.option_c
		ai_chance = { factor = 1 }
		trigger = {
			event_target:war_adventurer_province = {
				NOT = { has_building = counting_house }
				NOT = { trade_goods = gold }
			}
		}
		event_target:war_adventurer_province = {
			if = {
				limit = {
					NOT = { has_building = counting_house }
					NOT = { trade_goods = gold }
				}
				add_building = counting_house
			}
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	after = {
		hidden_effect = {
			event_target:war_adventurer_province = {
				if = {
					limit = { has_province_modifier = war_adventurer_bugman }
					war_adventurer_move = { MODIFIER = war_adventurer_bugman LOCATION = war_adventurer_location_bugman BEER = yes ARRIVAL_EVENT = war_adventurer_bugman.21 }
				}
			}
		}
	}
}
