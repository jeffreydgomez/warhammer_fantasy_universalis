country_event = {

		    owned_by = ROOT

		hidden_effect = {
			event_target:war_adventurer_location_gotrek_felix = {
				save_event_target_as = war_adventurer_province
				war_adventurer_prepare_destination = yes
			}
		}
	}
	goto = war_adventurer_province

	# Replace the abusive collection with verified local assessments. Option Weighing Value: +1.
	option = {
		name = war_adventurer_gotrek_felix.1.option_a
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_tax = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Return confiscated goods to the provincial shopkeepers. Option Weighing Value: +1.
	option = {

		ai_chance = { factor = 1 }

			add_base_production = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# File Felix's witness account without intervening. Option Weighing Value: +0.
	option = {
		name = war_adventurer_gotrek_felix.1.option_c
		ai_chance = { factor = 1 }
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
