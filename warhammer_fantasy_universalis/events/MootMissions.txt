# Mission choices for the Moot. These are mission rewards, not random events.
# Each choice grants 5 Weighing Value, already counted in its mission's budget.
namespace = moot_missions
normal_or_historical_nations = yes

# Witch Hunters at Supper: choose how this milestone shapes Mootland institutions.
country_event = {
	id = moot_missions.1
	title = moot_missions.1.t
	desc = moot_missions.1.d
	picture = EMPIRE_eventPicture
	is_triggered_only = yes
	fire_only_once = yes
	trigger = {
		is_or_was_tag = { tag = A14 }
		NOT = { has_country_flag = moot_mission_choice_1 }
	}
	immediate = { set_country_flag = moot_mission_choice_1 }

	# Protect our hearth customs. Weighing Value: 5.
	option = {
		name = moot_missions.1.a
		ai_chance = { factor = 50 }
		add_country_modifier = { name = moot_protected_hearths duration = -1 }
	}

	# License a watch answerable to the assembly. Weighing Value: 5.
	option = {
		name = moot_missions.1.b
		ai_chance = { factor = 50 }
		add_country_modifier = { name = moot_licensed_watch duration = -1 }
	}
}
