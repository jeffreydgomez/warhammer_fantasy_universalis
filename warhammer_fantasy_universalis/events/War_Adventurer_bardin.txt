# BAR-FH01: The Old Ironbreaker's Tunnel. Intended event Weighing Value: +1. Lore: L1.
country_event = {
	id = war_adventurer_bardin.1
	title = war_adventurer_bardin.1.t
	desc = war_adventurer_bardin.1.d
	picture = WAR_ADVENTURER_BAR_eventPicture
	is_triggered_only = yes
	trigger = {
		event_target:war_adventurer_location_bardin = {
		    owned_by = ROOT
		    has_province_modifier = war_adventurer_bardin
		}
	}
	immediate = {
		hidden_effect = {
			event_target:war_adventurer_location_bardin = {
				save_event_target_as = war_adventurer_province
				war_adventurer_prepare_destination = yes
			}
		}
	}
	goto = war_adventurer_province

	# Reopen the refuge as a permanent provincial shelter. Option Weighing Value: +1.
	option = {
		name = war_adventurer_bardin.1.option_a
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_tax = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Organize local rescue crews around the saved workers. Option Weighing Value: +1.
	option = {
		name = war_adventurer_bardin.1.option_b
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_manpower = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Establish training fields for the tunnel rescue crews. Option Weighing Value: +1.
	option = {
		name = war_adventurer_bardin.1.option_c
		ai_chance = { factor = 1 }
		trigger = {
			event_target:war_adventurer_province = {
				NOT = { has_building = training_fields }
			}
		}
		event_target:war_adventurer_province = {
			if = {
				limit = {
					NOT = { has_building = training_fields }
				}
				add_building = training_fields
			}
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	after = {
		hidden_effect = {
			event_target:war_adventurer_province = {
				if = {
					limit = { has_province_modifier = war_adventurer_bardin }
					war_adventurer_move = { MODIFIER = war_adventurer_bardin LOCATION = war_adventurer_location_bardin BEER = no ARRIVAL_EVENT = war_adventurer_bardin.21 }
				}
			}
		}
	}
}
