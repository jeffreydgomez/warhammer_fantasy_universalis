	id = war_adventurer_gelt.20
	title = war_adventurer_gelt.20.t
	desc = war_adventurer_gelt.20.d
	picture = WAR_ADVENTURER_GEL_eventPicture
	is_triggered_only = yes
	trigger = {
		event_target:war_adventurer_location_gelt = {
		    owned_by = ROOT
		    has_province_modifier = war_adventurer_gelt
		    war_adventurer_ready = yes
		    war_adventurer_gelt_home = no
		    owner = { war_adventurer_enemy_a = yes }
		    owner = {
		        any_country = {
		            NOT = { tag = ROOT }
		            any_owned_province = { war_adventurer_destination_province = yes }
		        }
		    }

		}
	}
	immediate = {
		event_target:war_adventurer_location_gelt = {
			save_event_target_as = war_adventurer_province
			set_province_flag = war_adventurer_event_open
		}
	}
	goto = war_adventurer_province

	# Exchange suspect coins at a loss. Option Weighing Value: -1.
	option = {
		name = war_adventurer_gelt.20.option_a
		ai_chance = { factor = 1 }
		add_years_of_income = -0.5
		custom_tooltip = war_adventurer_leaves_tt
	}

	# Impose a temporary accounting settlement. Option Weighing Value: -1.
	option = {
		name = war_adventurer_gelt.20.option_b
		ai_chance = { factor = 1 }
		add_adm_power = -50
		custom_tooltip = war_adventurer_leaves_tt
	}

	# Threaten the guilds. Option Weighing Value: -1.
	option = {
		name = war_adventurer_gelt.20.option_c
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_devastation = 15
		}
		add_prestige = -10
		custom_tooltip = war_adventurer_leaves_tt
	}

	after = {
		hidden_effect = {
			event_target:war_adventurer_province = {
				clr_province_flag = war_adventurer_event_open
				if = {
					limit = { has_province_modifier = war_adventurer_gelt }
					war_adventurer_move = { MODIFIER = war_adventurer_gelt LOCATION = war_adventurer_location_gelt BEER = no ARRIVAL_EVENT = war_adventurer_gelt.21 }
				}
			}
		}
	}
}
