# GEL-FH01: Powder Without Smoke. Intended event Weighing Value: +1. Lore: L2.
country_event = {
	id = war_adventurer_gelt.1
	title = war_adventurer_gelt.1.t
	desc = war_adventurer_gelt.1.d
	picture = WAR_ADVENTURER_GEL_eventPicture
	is_triggered_only = yes
	trigger = {
		event_target:war_adventurer_location_gelt = {
		    owned_by = ROOT
		    has_province_modifier = war_adventurer_gelt
		}
	}
	immediate = {
		hidden_effect = {
			event_target:war_adventurer_location_gelt = {
				save_event_target_as = war_adventurer_province
				war_adventurer_prepare_destination = yes
			}
		}
	}
	goto = war_adventurer_province

	# Issue the cleaner powder to the provincial militia. Option Weighing Value: +1.
	option = {
		name = war_adventurer_gelt.1.option_a
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_manpower = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Share the measurements with local furnace masters. Option Weighing Value: +1.
	option = {
		name = war_adventurer_gelt.1.option_b
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_production = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Endow a counting house for the powder guild. Option Weighing Value: +1.
	option = {
		name = war_adventurer_gelt.1.option_c
		ai_chance = { factor = 1 }
		trigger = {
			event_target:war_adventurer_province = {
				NOT = { has_building = counting_house }
				NOT = { trade_goods = gold }
			}
		}
		event_target:war_adventurer_province = {
			if = {
				limit = {
					NOT = { has_building = counting_house }
					NOT = { trade_goods = gold }
				}
				add_building = counting_house
			}
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	after = {
		hidden_effect = {
			event_target:war_adventurer_province = {
				if = {
					limit = { has_province_modifier = war_adventurer_gelt }
					war_adventurer_move = { MODIFIER = war_adventurer_gelt LOCATION = war_adventurer_location_gelt BEER = no ARRIVAL_EVENT = war_adventurer_gelt.21 }
				}
			}
		}
	}
}
