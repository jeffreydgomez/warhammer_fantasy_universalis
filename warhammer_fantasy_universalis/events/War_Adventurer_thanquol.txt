# THA-FH01: The Triumph in the Wrong Sewer. Intended event Weighing Value: +1. Lore: L1.
country_event = {
	id = war_adventurer_thanquol.1
	title = war_adventurer_thanquol.1.t
	desc = war_adventurer_thanquol.1.d
	picture = WAR_ADVENTURER_THA_eventPicture
	is_triggered_only = yes
	trigger = {
		event_target:war_adventurer_location_thanquol = {
		    owned_by = ROOT
		    has_province_modifier = war_adventurer_thanquol
		    war_adventurer_thanquol_home = yes
		    owner = { war_adventurer_enemy_b = no }

		}
	}
	immediate = {
		hidden_effect = {
			event_target:war_adventurer_location_thanquol = {
				save_event_target_as = war_adventurer_province
				war_adventurer_prepare_destination = yes
			}
		}
	}
	goto = war_adventurer_province

	# Distribute the recovered treasury goods to provincial workshops. Option Weighing Value: +1.
	option = {
		name = war_adventurer_thanquol.1.option_a
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_production = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Restore the local accounts exposed by the theft. Option Weighing Value: +1.
	option = {
		name = war_adventurer_thanquol.1.option_b
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_tax = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Equip provincial inspectors to guard the recovered stores. Option Weighing Value: +1.
	option = {
		name = war_adventurer_thanquol.1.option_c
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_manpower = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	after = {
		hidden_effect = {
			event_target:war_adventurer_province = {
				if = {
					limit = { has_province_modifier = war_adventurer_thanquol }
					war_adventurer_move = { MODIFIER = war_adventurer_thanquol LOCATION = war_adventurer_location_thanquol BEER = no ARRIVAL_EVENT = war_adventurer_thanquol.21 }
				}
			}
		}
	}
