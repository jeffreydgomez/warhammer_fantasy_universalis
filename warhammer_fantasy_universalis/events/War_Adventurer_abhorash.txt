# ABH-FH01: The Palace Steps Remember. Intended event Weighing Value: +1. Lore: L1.
country_event = {
	id = war_adventurer_abhorash.1
	title = war_adventurer_abhorash.1.t
	desc = war_adventurer_abhorash.1.d
	picture = WAR_ADVENTURER_ABH_eventPicture
	is_triggered_only = yes
	trigger = {
		event_target:war_adventurer_location_abhorash = {
		    owned_by = ROOT
		    has_province_modifier = war_adventurer_abhorash
		    war_adventurer_abhorash_home = yes
		    owner = { war_adventurer_enemy_d = no }

		}
	}
	immediate = {
		hidden_effect = {
			event_target:war_adventurer_location_abhorash = {
				save_event_target_as = war_adventurer_province
				war_adventurer_prepare_destination = yes
			}
		}
	}
	goto = war_adventurer_province

	# Preserve the corrected account beside the old inscription. Option Weighing Value: +1.
	option = {
		name = war_adventurer_abhorash.1.option_a
		ai_chance = { factor = 1 }
		change_innovativeness = 1
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Restore homes for the rescued palace workers. Option Weighing Value: +1.
	option = {
		name = war_adventurer_abhorash.1.option_b
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_tax = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Establish a university archive for the surviving palace records. Option Weighing Value: +1.
	option = {
		name = war_adventurer_abhorash.1.option_c
		ai_chance = { factor = 1 }
		trigger = {
			event_target:war_adventurer_province = {
				NOT = { has_building = university }
			}
		}
		event_target:war_adventurer_province = {
			if = {
				limit = {
					NOT = { has_building = university }
				}
				add_building = university
			}
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	after = {
		hidden_effect = {
			event_target:war_adventurer_province = {
				if = {
					limit = { has_province_modifier = war_adventurer_abhorash }
					war_adventurer_move = { MODIFIER = war_adventurer_abhorash LOCATION = war_adventurer_location_abhorash BEER = no ARRIVAL_EVENT = war_adventurer_abhorash.21 }
				}
			}
		}
	}
}
