# SNI-FH01: The Rival's Prayer. Intended event Weighing Value: +1. Lore: L1.
country_event = {
	id = war_adventurer_snikch.1
	title = war_adventurer_snikch.1.t
	desc = war_adventurer_snikch.1.d
	picture = WAR_ADVENTURER_SNI_eventPicture
	is_triggered_only = yes
	trigger = {
		event_target:war_adventurer_location_snikch = {
		    owned_by = ROOT
		    has_province_modifier = war_adventurer_snikch
		    owner = { war_adventurer_enemy_b = no }

		}
	}
	immediate = {
		hidden_effect = {
			event_target:war_adventurer_location_snikch = {
				save_event_target_as = war_adventurer_province
				war_adventurer_prepare_destination = yes
			}
		}
	}
	goto = war_adventurer_province

	# Bring the exposed payment route under provincial customs. Option Weighing Value: +1.
	option = {
		name = war_adventurer_snikch.1.option_a
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_tax = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Turn the seized workshops over to local suppliers. Option Weighing Value: +1.
	option = {
		name = war_adventurer_snikch.1.option_b
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_production = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Consolidate the seized trade in a stock exchange. Option Weighing Value: +1.
	option = {
		name = war_adventurer_snikch.1.option_c
		ai_chance = { factor = 1 }
		trigger = {
			event_target:war_adventurer_province = {
				NOT = { has_building = stock_exchange }
			}
		}
		event_target:war_adventurer_province = {
			if = {
				limit = {
					NOT = { has_building = stock_exchange }
				}
				add_building = stock_exchange
			}
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	after = {
		hidden_effect = {
			event_target:war_adventurer_province = {
				if = {
					limit = { has_province_modifier = war_adventurer_snikch }
					war_adventurer_move = { MODIFIER = war_adventurer_snikch LOCATION = war_adventurer_location_snikch BEER = no ARRIVAL_EVENT = war_adventurer_snikch.21 }
				}
			}
		}
	}
}
