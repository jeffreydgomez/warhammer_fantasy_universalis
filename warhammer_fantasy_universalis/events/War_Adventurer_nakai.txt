# NAK-FH01: The Crossing Before the Alarm. Intended event Weighing Value: +1. Lore: L1.
country_event = {
	id = war_adventurer_nakai.1
	title = war_adventurer_nakai.1.t
	desc = war_adventurer_nakai.1.d
	picture = WAR_ADVENTURER_NAK_eventPicture
	is_triggered_only = yes
	trigger = {
		event_target:war_adventurer_location_nakai = {
		    owned_by = ROOT
		    has_province_modifier = war_adventurer_nakai
		    war_adventurer_nakai_home = yes
		    owner = { war_adventurer_enemy_a = no }

		}
	}
	immediate = {
		hidden_effect = {
			event_target:war_adventurer_location_nakai = {
				save_event_target_as = war_adventurer_province
				war_adventurer_prepare_destination = yes
			}
		}
	}
	goto = war_adventurer_province

	# Rebuild the crossing and its provincial waystation. Option Weighing Value: +1.
	option = {
		name = war_adventurer_nakai.1.option_a
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_tax = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Train local crossing guards from the battle account. Option Weighing Value: +1.
	option = {
		name = war_adventurer_nakai.1.option_b
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_manpower = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Restore provincial workshops along the reopened supply route. Option Weighing Value: +1.
	option = {
		name = war_adventurer_nakai.1.option_c
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_production = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	after = {
		hidden_effect = {
			event_target:war_adventurer_province = {
				if = {
					limit = { has_province_modifier = war_adventurer_nakai }
					war_adventurer_move = { MODIFIER = war_adventurer_nakai LOCATION = war_adventurer_location_nakai BEER = no ARRIVAL_EVENT = war_adventurer_nakai.21 }
