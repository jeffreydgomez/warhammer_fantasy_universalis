# SIE-FH01: Thyrus's Marginal Notes. Intended event Weighing Value: +1. Lore: L1.
country_event = {
	id = war_adventurer_sienna.1
	title = war_adventurer_sienna.1.t
	desc = war_adventurer_sienna.1.d
	picture = WAR_ADVENTURER_SIE_eventPicture
	is_triggered_only = yes
	trigger = {
		event_target:war_adventurer_location_sienna = {
		    owned_by = ROOT
		    has_province_modifier = war_adventurer_sienna
		}
	}
	immediate = {
		hidden_effect = {
			event_target:war_adventurer_location_sienna = {
				save_event_target_as = war_adventurer_province
				war_adventurer_prepare_destination = yes
			}
		}
	}
	goto = war_adventurer_province

	# Record the corrected furnace design. Option Weighing Value: +1.
	option = {
		name = war_adventurer_sienna.1.option_a
		ai_chance = { factor = 1 }
		change_innovativeness = 1
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Rebuild local furnaces with the safer air channels. Option Weighing Value: +1.
	option = {
		name = war_adventurer_sienna.1.option_b
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_production = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Equip a counting house for the improved metalworking trade. Option Weighing Value: +1.
	option = {
		name = war_adventurer_sienna.1.option_c
		ai_chance = { factor = 1 }
		trigger = {
			event_target:war_adventurer_province = {
				NOT = { has_building = counting_house }
				NOT = { trade_goods = gold }
			}
		}
		event_target:war_adventurer_province = {
			if = {
				limit = {
					NOT = { has_building = counting_house }
					NOT = { trade_goods = gold }
				}
				add_building = counting_house
			}
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	after = {
		hidden_effect = {
			event_target:war_adventurer_province = {
				if = {
					limit = { has_province_modifier = war_adventurer_sienna }
					war_adventurer_move = { MODIFIER = war_adventurer_sienna LOCATION = war_adventurer_location_sienna BEER = no ARRIVAL_EVENT = war_adventurer_sienna.21 }
				}
			}
		}
	}
}
