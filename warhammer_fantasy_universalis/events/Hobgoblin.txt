	desc = hobgoblin_events.1000.d
	picture = HOBGOBLIN_eventPicture

	fire_only_once = no
	is_triggered_only = yes

	trigger = {
		culture_group = hobgoblin_group

		create_general = { tradition = 100 }
		add_yearly_manpower = 1
		add_mil_power = 50
		add_years_of_income = 0.5
	}
}
