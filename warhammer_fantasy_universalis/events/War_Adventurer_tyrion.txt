# TYR-FH01: The Shrine's Renewed Watch. Intended event Weighing Value: +1. Lore: L1.
country_event = {
	id = war_adventurer_tyrion.1
	title = war_adventurer_tyrion.1.t
	desc = war_adventurer_tyrion.1.d
	picture = WAR_ADVENTURER_TYR_eventPicture
	is_triggered_only = yes
	trigger = {
		event_target:war_adventurer_location_tyrion = {
		    owned_by = ROOT
		    has_province_modifier = war_adventurer_tyrion
		    war_adventurer_tyrion_home = yes
		    owner = { war_adventurer_enemy_a = no }

		}
	}
	immediate = {
		hidden_effect = {
			event_target:war_adventurer_location_tyrion = {
				save_event_target_as = war_adventurer_province
				war_adventurer_prepare_destination = yes
			}
		}
	}
	goto = war_adventurer_province

	# Restore the sanctuary precinct with the recovered offerings. Option Weighing Value: +1.
	option = {
		name = war_adventurer_tyrion.1.option_a
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_tax = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Train the provincial sanctuary watch. Option Weighing Value: +1.
	option = {
		name = war_adventurer_tyrion.1.option_b
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_manpower = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Raise a cathedral around the restored sanctuary. Option Weighing Value: +1.
	option = {
		name = war_adventurer_tyrion.1.option_c
		ai_chance = { factor = 1 }
		trigger = {
			event_target:war_adventurer_province = {
				NOT = { has_building = cathedral }
			}
		}
		event_target:war_adventurer_province = {
			if = {
				limit = {
					NOT = { has_building = cathedral }
				}
				add_building = cathedral
			}
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	after = {
		hidden_effect = {
			event_target:war_adventurer_province = {
				if = {
					limit = { has_province_modifier = war_adventurer_tyrion }
					war_adventurer_move = { MODIFIER = war_adventurer_tyrion LOCATION = war_adventurer_location_tyrion BEER = no ARRIVAL_EVENT = war_adventurer_tyrion.21 }
				}
			}
		}
	}
}
