# ELS-FH01: The Tower's Returned Reliquary. Intended event Weighing Value: +1. Lore: L1.
country_event = {
	id = war_adventurer_elspeth.1
	title = war_adventurer_elspeth.1.t
	desc = war_adventurer_elspeth.1.d
	picture = WAR_ADVENTURER_ELS_eventPicture
	is_triggered_only = yes
	trigger = {
		event_target:war_adventurer_location_elspeth = {
		    owned_by = ROOT
		    has_province_modifier = war_adventurer_elspeth
		    owner = { war_adventurer_enemy_a = no }

		}
	}
	immediate = {
		hidden_effect = {
			event_target:war_adventurer_location_elspeth = {
				save_event_target_as = war_adventurer_province
				war_adventurer_prepare_destination = yes
			}
		}
	}
	goto = war_adventurer_province

	# Reopen the corrected burial precinct and its local records. Option Weighing Value: +1.
	option = {
		name = war_adventurer_elspeth.1.option_a
		ai_chance = { factor = 1 }
		event_target:war_adventurer_province = {
			add_base_tax = 2
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# Have scholars examine the corrected funerary inscription. Option Weighing Value: +1.
	option = {
		name = war_adventurer_elspeth.1.option_b
		ai_chance = { factor = 1 }
		change_innovativeness = 1
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	# House the restored reliquary in a new cathedral. Option Weighing Value: +1.
	option = {
		name = war_adventurer_elspeth.1.option_c
		ai_chance = { factor = 1 }
		trigger = {
			event_target:war_adventurer_province = {
				NOT = { has_building = cathedral }
			}
		}
		event_target:war_adventurer_province = {
			if = {
				limit = {
					NOT = { has_building = cathedral }
				}
				add_building = cathedral
			}
		}
		if = {
			limit = {
				event_target:war_adventurer_destination_country = { war_adventurer_destination_country = yes }
			}
			custom_tooltip = war_adventurer_leaves_tt
		}
		else = { custom_tooltip = war_adventurer_stays_tt }
	}

	after = {
		hidden_effect = {
			event_target:war_adventurer_province = {
				if = {
					limit = { has_province_modifier = war_adventurer_elspeth }
					war_adventurer_move = { MODIFIER = war_adventurer_elspeth LOCATION = war_adventurer_location_elspeth BEER = no ARRIVAL_EVENT = war_adventurer_elspeth.21 }
				}
			}
		}
	}
}
