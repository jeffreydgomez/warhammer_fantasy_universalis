country_event = {
	id = war_formable_nations.1
	title = war_formable_nations.1.t
	desc = war_formable_nations.1.d
	picture = KISLEV_eventPicture
	is_triggered_only = yes

	# Adopt the new national ideas.
	option = {
		name = ideagroups.1.a
		ai_chance = { factor = 100 }
		trigger = { has_custom_ideas = no }
		hidden_effect = {
			if = {
				limit = {
					has_idea_group = norse_ideas
				}
				set_country_flag = refused_norse_ideas
			}
		}
		swap_free_idea_group = yes
	}
	# Keep our current national ideas.
	option = {
		name = ideagroups.1.b
		ai_chance = { factor = 0 }
	}
}
