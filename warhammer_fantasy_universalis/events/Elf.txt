### Ulthuan for the Asur

country_event = { 
	id = asur_events.0
	title = asur_events.0.t 
	desc = asur_events.0.d 
	picture = HIGHELF_eventPicture
	
	fire_only_once = no
	is_triggered_only = no 
		
	trigger = {
		culture_group = asur_group
		religion = cadai
		is_at_war = no
		capital_scope = { superregion = ulthuan_superregion }
		any_country = {
			NOT = { culture_group = asur_group }
			NOT = { truce_with = ROOT }
			ROOT = {
				core_claim = PREV
			}
			ROOT = {
				NOT = {
					has_opinion = {
						who = PREV
						value = 0
					}
				}
			}
			provinces_on_capital_continent_of = ROOT
		}
	}

	mean_time_to_happen = {
		months = 160
	}


# Declare War
	option = { 
		ai_chance = {
			factor = 1
			modifier = {
				factor = 100
				is_in_coalition = yes
				manpower_percentage = 0.50
			}
		}
		name = asur_events.0.a 
		random_country = {
			limit = {
				NOT = { culture_group = asur_group }
				NOT = { truce_with = ROOT }
				ROOT = {
					NOT = {
						has_opinion = {
							who = PREV
							value = 0
						}
					}
				}
				provinces_on_capital_continent_of = ROOT
			}
			ROOT = {
				declare_war_with_cb = {
				    who = PREV
				    casus_belli = cb_super_badboy
				}
			}
		}
		add_prestige = 25
	}

## Let us gather a coalition
	option = { 
		ai_chance = {
			factor = 100
			modifier = {
				factor = 10
				OR = {
					NOT = { manpower_percentage = 0.10 }
					is_at_war = yes
					num_of_loans = 10
					is_bankrupt = yes
					NOT = { stability = -1 }
				}
			}
		}
		name = asur_events.0.b 

		add_prestige = -10
		add_legitimacy = -5
		add_devotion = -5
		add_republican_tradition = -2


		random_country = {
			limit = {
				NOT = { culture_group = asur_group }
				NOT = { truce_with = ROOT }
				ROOT = {
					NOT = {
						has_opinion = {
							who = PREV
							value = 0
						}
					}
				}
				provinces_on_capital_continent_of = ROOT
			}
			
			every_country = {
				limit = {
					culture_group = asur_group
					religion = cadai
					capital_scope = { superregion = ulthuan_superregion }
					NOT = {
						OR = {
							alliance_with = PREV
							is_subject_of = PREV
						}
					}
				}

				add_aggressive_expansion = {
					who = PREV
					value = 10
					apply_calc = no
				}
			}
		}
	}



## Submit to Naggaroth
	option = { 
		ai_chance = {
			factor = 1
			modifier = {
				factor = 50
				H12 = { religion = cytharai }
				religion = cytharai
			}
			modifier = {
				factor = 50
				H12 = { 
					ROOT = {
						has_opinion = {
							who = PREV
							value = 150
						}
					}
				}
			}
		}
		name = asur_events.0.c
		trigger = {
			H12 = {
				provinces_on_capital_continent_of = ROOT
				NOT = {
					ROOT = {
