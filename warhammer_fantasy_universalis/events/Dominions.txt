### Trade Favors for helper Event
country_event = {
	id = dominion_events.2
	title = dominion_events.2.t
	desc = dominion_events.2.d
	picture = ARTWORK_eventPicture

	trigger = {
		num_of_subjects = 1
	}

	is_triggered_only = no
	hidden = yes
	mean_time_to_happen = { months = 6 }
	
	option = {
		name = dominion_events.2.option_1
		ai_chance = { factor = 100 }

		every_subject_country = {
			limit = { NOT = { liberty_desire = 50 } }
			ROOT = {
				add_favors = {
			   		who = PREV
			   		amount = 1
				}
			}

			add_trust = {
			    who = ROOT
			    value = 1
			}
		}

		every_subject_country = {
			limit  = {
				NOT = {
					trust = {
						who = ROOT
						value = 100
					}
				}

				owes_favors = { 
				    who = ROOT 
				    value = 100 
				}
			}

			ROOT = {
				add_favors = {
			   		who = PREV
			   		amount = -10
				}
			}

			add_trust = {
			    who = ROOT
			    value = 5
			}
		}
	}

}
