# Concept: execute sabotage without exposing native tier changes to hover tooltips.
country_event = {
	id = war_warpmoon.10
	title = war_destroy_warpmoon_project_title
	desc = war_destroy_warpmoon_project_desc
	picture = WAR_WARPMOON_CONSTRUCTION_eventPicture
	hidden = yes
	is_triggered_only = yes
	immediate = {
		4600 = {
			if = {
				limit = { controlled_by = ROOT }
				# Native project effects live directly in this event, not in a
				# scripted-effect expansion used by the decision's description.
				# Exact-tier branches leave the existing project at tier 0.
				if = {
					limit = { has_great_project = { type = war_warpmoon_project tier = 2 } }
					add_great_project_tier = { type = war_warpmoon_project tier = -2 }
				}
				else_if = {
					limit = { has_great_project = { type = war_warpmoon_project tier = 1 } }
					add_great_project_tier = { type = war_warpmoon_project tier = -1 }
				}
			}
		}
	}
	# Finish the hidden helper; all work is resolved in immediate.
	option = { name = war_warpmoon_internal.a }
}

# Concept: fire the completed cannon once, then report triumph to its owner and ruin abroad.
province_event = {
	id = war_warpmoon.13
	title = war_warpmoon.6.t
	desc = war_warpmoon.6.d
	picture = WAR_WARPMOON_eventPicture
	hidden = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = { NOT = { has_global_flag = war_warpmoon_moon_destroyed } }
			set_global_flag = war_warpmoon_moon_destroyed
			save_event_target_as = war_warpmoon_province
			owner = {
				save_event_target_as = war_warpmoon_builder
			}
			4600 = { destroy_great_project = { type = war_warpmoon_project } }

			every_province = {
				limit = { war_warpmoon_impact_eligible = yes }
				random = {
					chance = 15
					war_warpmoon_impact = yes
				}
			}
			every_country = {
				limit = {
					NOT = { has_country_modifier = skaven_administration }
					NOT = { tag = event_target:war_warpmoon_builder }
				}
				# Recipients gain AE against the builder. User-specified, unpriced
				# in the Balance Guide; no distance, religion or AE scaling.
				add_aggressive_expansion = {
					who = event_target:war_warpmoon_builder
					value = 50
					apply_calc = no
				}
			}
			owner = { country_event = { id = war_warpmoon.6 } }
			every_country = {
				limit = { NOT = { tag = event_target:war_warpmoon_builder } }
				country_event = { id = war_warpmoon.3 }
			}
		}
	}
	# Finish the hidden helper; all work is resolved in immediate.
	option = { name = war_warpmoon_internal.a }
}
