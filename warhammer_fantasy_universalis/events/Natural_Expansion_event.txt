########################################
# Events for Natural Expansion
########################################

namespace = natural_expansion_event
normal_or_historical_nations = yes


country_event = { 
	id = natural_expansion_event.0
	title = natural_expansion_event.0.t 
	desc = natural_expansion_event.0.d 
	picture = IMPORTANT_STATUE_eventPicture
	
	fire_only_once = no
	is_triggered_only = yes 
		
	trigger = {
		ai = yes
	}

	mean_time_to_happen = { days = 1 }

	option = { 
		ai_chance = { factor = 100 }
		name = natural_expansion_event.0.a 
		war_natural_expansion = yes


		# AI Prophet of the Sun slowly eats AI tributaries
		if = {
			limit = {
				is_emperor_of_china = yes
				is_at_war = no
			}
			every_subject_country = {
				limit = {
					ai = yes
					is_subject_of_type = tributary_state
					is_at_war = no		
					has_opinion = {
						who = ROOT
						value = 190
					}
				}
				random_owned_province = {
					limit = {
						OR = {
							religion = ROOT
							culture_group = ROOT
						}
						any_neighbor_province = {
							owned_by = ROOT
						}
						NOT = {
							any_neighbor_province = {
								owner = {
									NOT = { is_subject_of = ROOT }
									NOT = { tag = ROOT }
#									total_development = 20
								}
							}
						}
					}
					cede_province = ROOT
					add_core = ROOT
				}
			}

			## Convert neighboring countries into tributaries
			every_neighbor_country = {
				limit = {
					ai = yes
					NOT = { is_subject = yes }
					NOT = { has_subject_of_type = tributary_state }
					NOT = { total_development = ROOT }
					NOT = { army_size = ROOT }
					OR = {
						is_threat = ROOT
						has_opinion = {
							who = ROOT
							value = 10
						}
					}
				}
				ROOT = {
					create_subject = {
					    subject_type = tributary_state
					    subject = PREV
					}
				}
			}
		}
	}
}
