### With the entire Nehekhara region following our culture, we can more closely train our troops and make sure our infantry is in lockstep with each other.
# Full idea
	nagashizzar_unify_culture = {
		icon = mission_mosque
		position = 9
		required_missions = {}
		provinces_to_highlight = {
			OR = {
				AND = {
					NOT = { owned_by = ROOT }
					has_great_project = {
						type = tomb_of_settra
						tier = 0
					}
				}
				AND = {
					region = nehekhara_region
					NOT = {
						culture_group = ROOT
					}
				}
			}
		}
		trigger = {

		# Performance
			hidden_trigger = {
				OR = {
					ai = no
					num_of_cities = 25
				}
			}


			NOT = {
				any_province = {
					region = nehekhara_region
					NOT = {
						culture_group = ROOT
					}
				}
			}
			any_owned_province = {
				has_great_project = {
					type = tomb_of_settra
					tier = 0
				}
			}
		}
		effect = {
			add_country_modifier = { 
    			name = nagashizzar_unify_culture
    			duration = -1  
			}
		}
	}
