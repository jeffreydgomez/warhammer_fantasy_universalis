		position = 5
		required_missions = { khuresh_older_than_plaques }
		provinces_to_highlight = {
			owned_by = ROOT
			OR = {
				region = north_khuresh_region
				region = south_khuresh_region
			}
			has_terrain = jungle
			is_prosperous = no
		}
		trigger = {
			any_owned_province = {
				OR = {
					region = north_khuresh_region
					region = south_khuresh_region
				}
				has_terrain = jungle
			}
			if = {
				limit = { has_dlc = "Mandate of Heaven" }
				OR = {
					all_owned_province = {
						is_prosperous = yes
					}
					num_of_owned_provinces_with = {
						is_prosperous = yes
						value = 12
					}
				}
			}
			else = {
				OR = {
					all_owned_province = {
						AND = {
							NOT = { devastation = 1 }
							development = 4
						}
					}
					num_of_owned_provinces_with = {
						NOT = { devastation = 1 }
						development = 4
						value = 12
					}
				}
			}
		}
		effect = {
			random_owned_province = {
				limit = {
					OR = {
						region = north_khuresh_region
						region = south_khuresh_region
					}
					has_terrain = jungle
					NOT = { has_province_modifier = khuresh_progenitor_root_sign }
				}
				add_permanent_province_modifier = {
					name = khuresh_progenitor_root_sign
					duration = -1
				}
				save_event_target_as = khuresh_progenitor_grove
			}
			country_event = { id = khuresh_events.102 }
		}
	}
