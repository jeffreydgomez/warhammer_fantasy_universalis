	resurrect_nagash_decision = {
		potential = {

			## Performance

			OR = {
				ai = no
				religion = vampiric
				religion = nagash_worship
			}

			NOT = { has_global_flag = nagash_resurrected }

			OR = {

				religion = vampiric
				religion = nagash_worship
				has_country_modifier = undead_administration

				any_owned_province = {
					has_province_modifier = location_of_the_jade_blooded_vial
				}
				any_owned_province = {
					has_province_modifier = location_of_the_staff_of_nagash
				}
				any_owned_province = {
					has_province_modifier = location_of_the_carstein_ring
				}
				any_owned_province = {
					has_province_modifier = location_of_nagashs_tome
				}
				any_owned_province = {
					has_province_modifier = location_of_the_bleeding_idol
				}
			}
		 }

		 provinces_to_highlight = {
			# Once four distinct artifacts are owned, the fifth is not an objective.
			ROOT = {
				NOT = {
					calc_true_if = {
						amount = 4
						AND = { any_owned_province = { has_province_modifier = location_of_the_jade_blooded_vial } }
						AND = { any_owned_province = { has_province_modifier = location_of_the_staff_of_nagash } }
						AND = { any_owned_province = { has_province_modifier = location_of_the_carstein_ring } }
						AND = { any_owned_province = { has_province_modifier = location_of_nagashs_tome } }
						AND = { any_owned_province = { has_province_modifier = location_of_the_bleeding_idol } }
					}
				}
			}
		 	OR = {
				AND = {
					has_province_modifier = location_of_the_jade_blooded_vial
					NOT = { owned_by = ROOT }
				}
				AND = {
					has_province_modifier = location_of_the_staff_of_nagash
					NOT = { owned_by = ROOT }
				}
				AND = {
					has_province_modifier = location_of_the_carstein_ring
					NOT = { owned_by = ROOT }
				}
				AND = {
					has_province_modifier = location_of_nagashs_tome
					NOT = { owned_by = ROOT }
				}
				AND = {
					has_province_modifier = location_of_the_bleeding_idol
					NOT = { owned_by = ROOT }
				}
			}
		}

		allow = {
			OR = {
				religion = vampiric
				religion = nagash_worship
			}

		manpower = 50
		treasury = 1000

			custom_trigger_tooltip = {
				tooltip = war_nagash_four_artifacts_tt
				calc_true_if = {
					amount = 4
					AND = { any_owned_province = { has_province_modifier = location_of_the_jade_blooded_vial } }
					AND = { any_owned_province = { has_province_modifier = location_of_the_staff_of_nagash } }
					AND = { any_owned_province = { has_province_modifier = location_of_the_carstein_ring } }
					AND = { any_owned_province = { has_province_modifier = location_of_nagashs_tome } }
					AND = { any_owned_province = { has_province_modifier = location_of_the_bleeding_idol } }
				}
			}

		}

		effect = {
			add_manpower = -50
			add_treasury = -1000
			custom_tooltip = war_nagash_consume_four_artifacts_tt
			hidden_effect = {
				# Check the original collection before consuming anything. The Idol is
				# needed only when one of the other four artifacts is missing.
				if = {
					limit = {
						NOT = {
							AND = {
								any_owned_province = { has_province_modifier = location_of_the_jade_blooded_vial }
								any_owned_province = { has_province_modifier = location_of_the_staff_of_nagash }
								any_owned_province = { has_province_modifier = location_of_the_carstein_ring }
								any_owned_province = { has_province_modifier = location_of_nagashs_tome }
							}
						}
					}
					every_owned_province = {
						limit = { has_province_modifier = location_of_the_bleeding_idol }
						remove_province_modifier = location_of_the_bleeding_idol
						clear_global_event_target = war_artifact_location_bleeding_idol
					}
				}
				# Consume only artifacts owned by the country performing the ritual.
				every_owned_province = {
					limit = { has_province_modifier = location_of_the_jade_blooded_vial }
					remove_province_modifier = location_of_the_jade_blooded_vial
					clear_global_event_target = war_artifact_location_jade_blooded_vial
				}
				every_owned_province = {
					limit = { has_province_modifier = location_of_the_staff_of_nagash }
					remove_province_modifier = location_of_the_staff_of_nagash
					clear_global_event_target = war_artifact_location_staff_of_nagash
				}
				every_owned_province = {
					limit = { has_province_modifier = location_of_the_carstein_ring }
					remove_province_modifier = location_of_the_carstein_ring
					clear_global_event_target = war_artifact_location_carstein_ring
				}

					remove_province_modifier = location_of_nagashs_tome
					clear_global_event_target = war_artifact_location_nagashs_tome
				}
			}

			add_country_modifier = { 
			    name = nagash_resurrected_modifier
			    duration = -1
			}
			save_global_event_target_as = war_artifact_nagash_holder

			country_event = { 
			    id = tomb_king.100

			set_global_flag = nagash_resurrected
